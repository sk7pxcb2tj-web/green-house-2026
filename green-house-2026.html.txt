        window.switchPage = (pageId) => {

        // Firestore Data Loaders & Actions
        window.loadDashboard = async () => {
            try {
                const shirtsSnap = await getDocs(collection(db, "shirtOrders"));
                const sponsSnap = await getDocs(collection(db, "sponsors"));
                const expSnap = await getDocs(collection(db, "expenses"));

                let shirtRev = 0, sponsRev = 0, spent = 0;
                shirtsSnap.forEach(d => shirtRev += Number(d.data().totalPrice || 0));
                sponsSnap.forEach(d => sponsRev += Number(d.data().amount || 0));
                expSnap.forEach(d => {
                    const data = d.data();
                    if (data.status === 'อนุมัติแล้ว') spent += Number(data.amount || 0);
                });

                document.getElementById('stat-shirt-rev').innerText = '฿' + shirtRev.toLocaleString();
                document.getElementById('stat-spons').innerText = '฿' + sponsRev.toLocaleString();
                document.getElementById('stat-spent').innerText = '฿' + spent.toLocaleString();
                document.getElementById('stat-balance').innerText = '฿' + ((shirtRev + sponsRev) - spent).toLocaleString();
            } catch (e) { console.error(e); }
        };

        window.loadShirtOrders = async () => {
            const tbody = document.getElementById('shirt-orders-tbody');
            const snap = await getDocs(collection(db, "shirtOrders"));
            if (snap.empty) { tbody.innerHTML = `<tr><td colspan="5" class="text-center py-6 text-gray-400">ยังไม่มีคำสั่งซื้อ</td></tr>`; return; }
            const isAdmin = currentUser && ['Super Admin', 'Financial Admin', 'Sub Admin'].includes(currentUser.role);
            let html = "";
            snap.forEach(docSnap => {
                const o = docSnap.data();
                const id = docSnap.id;
                html += `
                    <tr class="hover:bg-gray-50 transition">
                        <td class="p-3 font-bold">${o.studentId}</td>
                        <td class="p-3">${o.name}</td>
                        <td class="p-3 font-semibold">${o.size} (${o.qty})</td>
                        <td class="p-3 text-emerald-600 font-bold">฿${o.totalPrice}</td>
                        <td class="p-3">
                            ${isAdmin ? `<select onchange="updateShirtStatus('${id}', this.value)" class="form-input-apple py-1 px-2 text-xs">
                                <option value="รอตรวจสอบ" ${o.status === 'รอตรวจสอบ' ? 'selected' : ''}>รอตรวจสอบ</option>
                                <option value="ชำระเงินแล้ว" ${o.status === 'ชำระเงินแล้ว' ? 'selected' : ''}>ชำระเงินแล้ว</option>
                                <option value="รับเสื้อแล้ว" ${o.status === 'รับเสื้อแล้ว' ? 'selected' : ''}>รับเสื้อแล้ว</option>

        window.claimMeal = async (dayNum) => {
            if (!checkAuthGuard()) return;
            const studentId = document.getElementById('meal-studentid').value.trim();
            if (!studentId) return alert('กรุณาระบุรหัสนักเรียน');
            
            const snap = await getDocs(collection(db, "meals"));
            let alreadyClaimed = false;
            snap.forEach(d => {
                const m = d.data();
                if (m.studentId === studentId && m.day === dayNum) alreadyClaimed = true;
            });

            if (alreadyClaimed) {
                Swal.fire('แจ้งเตือน', `รหัสนักเรียน ${studentId} ได้รับอาหารวันที่ ${dayNum} ไปแล้ว!`, 'warning');
                return;
            }

            await addDoc(collection(db, "meals"), { studentId, day: dayNum, timestamp: new Date() });
            showToast(`บันทึกรับอาหาร วันที่ ${dayNum} สำเร็จ!`);
            document.getElementById('meal-studentid').value = '';
        };
